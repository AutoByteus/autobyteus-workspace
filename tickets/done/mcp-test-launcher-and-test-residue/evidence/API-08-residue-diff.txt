#!/bin/bash
# API-08: AC-005 / AC-006 / REQ-004 / REQ-005 / REQ-006 residue run on the final commit.
set -u
W=/Users/normy/autobyteus_org/autobyteus-worktrees/mcp-test-launcher-and-test-residue
O=/tmp/abyt199-api/api08
UVCACHE=/var/folders/7w/9r4_s1_s42z3f7c136bpjf0r0000gn/T/autobyteus-mcp-test-server-uv-cache
PREFIXES='^(mcp-config-|tmp-edit-file-|tmp-read-file-|tmp-write-file-|agent-runtime-memory-|file-explorer-graphql-appdata-|path-boundary-e2e-appdata-|file-explorer-ws-e2e-appdata-|file-ops-graphql-appdata-|workspaces-graphql-appdata-)'
cd "$W"
snap() { git status --ignored --porcelain --untracked-files=all | grep -v node_modules | grep -v '^?? tickets/' | sort; }
: > $O/exits.txt
echo "commit $(git rev-parse --short HEAD)" >> $O/exits.txt
s=$(date +%s); pnpm -C autobyteus-server-ts build > $O/build.log 2>&1; echo "build exit $? ($(( $(date +%s)-s ))s)" >> $O/exits.txt
snap > $O/snap-0-before.txt
suite() { # name, then command words
  local name=$1; shift
  local TD H s
  TD="$(cd "$(mktemp -d /tmp/a8.XXXXXX)" && pwd -P)"; H="$(cd "$(mktemp -d)" && pwd -P)"
  echo "$name TMPDIR=$TD HOME=$H" >> $O/exits.txt
  s=$(date +%s)
  env -i PATH="$PATH" HOME="$H" TMPDIR="$TD" UV_CACHE_DIR="$UVCACHE" "$@" > $O/$name.log 2>&1
  echo "$name exit $? ($(( $(date +%s)-s ))s)" >> $O/exits.txt
  ls -A "$TD" | sort > $O/tmpdir-left-$name.txt
  echo "$name tmpdir entries left: $(wc -l < $O/tmpdir-left-$name.txt | tr -d ' '); touched-prefix entries left: $(grep -cE "$PREFIXES" $O/tmpdir-left-$name.txt)" >> $O/exits.txt
  rm -rf "$TD" "$H"
}
suite core pnpm -C autobyteus-ts test
snap > $O/snap-1-after-core.txt
suite server-unit pnpm -C autobyteus-server-ts test:unit
snap > $O/snap-2-after-unit.txt
suite e2e pnpm test:e2e
snap > $O/snap-3-after-e2e.txt
for i in 1-after-core 2-after-unit 3-after-e2e; do diff $O/snap-0-before.txt $O/snap-$i.txt > $O/diff-$i.txt; echo "diff before vs $i exit $? ($(wc -l < $O/diff-$i.txt | tr -d ' ') lines)" >> $O/exits.txt; done
echo DONE >> $O/exits.txt
