def hc(p):
    d=json.load(open(p)); out={}
    for a in d['advisories'].values():
        if a['severity'] not in ('critical','high'): continue
        for f in a.get('findings',[]):
            k=f"{a['module_name']}@{f['version']}"; out.setdefault(k,set()).add((a['severity'],a.get('github_advisory_id') or a.get('url','').rsplit('/',1)[-1],a.get('created','')[:10],a.get('patched_versions')))
    return d['metadata']['vulnerabilities'],out
