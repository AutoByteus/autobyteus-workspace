# Generates a local runner script for the build-linux-arm64 job of release-desktop.yml (build-only choices).
import sys, yaml, re
wt, commit, root, out = sys.argv[1:5]
w = yaml.safe_load(open(f"{wt}/.github/workflows/release-desktop.yml"))
steps = w["jobs"]["build-linux-arm64"]["steps"]
L = ["set -euo pipefail", "export PATH=/tmp/sm-bin:$PATH", "export GOTOOLCHAIN=local",
     "export PATH=$(go env GOMODCACHE 2>/dev/null || echo /root/go/pkg/mod)/golang.org/toolchain@v0.0.1-go1.27.4.linux-arm64/bin:$PATH",
     f"ROOT={root}", "rm -rf $ROOT && mkdir -p $ROOT/runner-temp",
     f"git clone --no-local {wt} $ROOT/clone", f"cd $ROOT/clone && git checkout --quiet {commit} && git log --oneline -1",
     "export RUNNER_TEMP=$ROOT/runner-temp GITHUB_ENV=$ROOT/github-env GITHUB_REPOSITORY=AutoByteus/autobyteus-workspace", "touch $GITHUB_ENV",
     "mkdir -p /tmp/oc-libz && ln -sf /usr/lib/aarch64-linux-gnu/libz.so.1 /tmp/oc-libz/libz.so && export LD_LIBRARY_PATH=/tmp/oc-libz"]
ENV_CHOICES = {"DESKTOP_RELEASE_PUBLIC_KEY": "", "DESKTOP_TRUST_MODE": "build-only", "AUTOBYTEUS_BUILD_FLAVOR": "personal"}
TOLERATED = {"Check native and signed-record units": "root-limited (uid 0 refused)", "Verify Linux ARM64 AppImage runs in a clean Ubuntu image": "needs docker (not installed here)"}
for st in steps:
    name = st.get("name")
    if "uses" in st: L.append(f"echo '== SKIP (action, emulated): {name}'"); continue
    if name == "Install Linux build deps": L.append(f"echo '== SKIP (host package install): {name}'"); continue
    env = st.get("env") or {}
    body = st["run"]
    if "${{" in body: raise SystemExit(f"expression in run of {name}")
    L.append(f"echo '== STEP: {name}'")
    wd = st.get("working-directory") or ((w["jobs"]["build-linux-arm64"].get("defaults") or {}).get("run") or {}).get("working-directory") or ((w.get("defaults") or {}).get("run") or {}).get("working-directory")
    if wd and "${{" in wd: raise SystemExit(f"expression in working-directory of {name}")
    L.append('( set -a; [ -s "$GITHUB_ENV" ] && . "$GITHUB_ENV"; set +a; cd $ROOT/clone' + (f"/{wd}" if wd else ""))
    for k in env:
        v = ENV_CHOICES.get(k)
        if v is None: raise SystemExit(f"no choice for env {k} in {name}")
        L.append(f"export {k}='{v}'")
    L.append(body)
    L.append(f") || {{ echo 'STEP FAILED (tolerated: {TOLERATED[name]}): {name}'; }}" if name in TOLERATED else ")")
L.append("echo FRESH_START_JOB_DONE")
open(out, "w").write("\n".join(L) + "\n")
