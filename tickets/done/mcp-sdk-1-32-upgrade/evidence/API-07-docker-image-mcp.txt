const startServer = async ({ marker, redirectTo }) => {
  const observations = [];
  const srv = http.createServer(async (req, res) => {
    let raw = ''; for await (const chunk of req) raw += chunk;
    const body = raw ? JSON.parse(raw) : undefined;
    const auth = req.headers.authorization === undefined ? 'none' : req.headers.authorization === `Bearer ${DUMMY}` ? 'expected' : 'other';
    observations.push({ method: req.method, rpc: body?.method ?? null, auth });
    if (redirectTo) { res.writeHead(307, { location: redirectTo() }).end(); return; }
    if (auth !== 'expected') { res.writeHead(401).end('unauthorized'); return; }
    const mcp = new McpServer({ name: 'packaged-probe', version: '1' });
    mcp.tool('auth_probe', 'Returns the marker.', async () => ({ content: [{ type: 'text', text: marker }] }));
    const t = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined });
    res.on('close', () => { void t.close(); void mcp.close(); });
    await mcp.connect(t); await t.handleRequest(req, res, body);
  });
  await new Promise((r) => srv.listen(0, '127.0.0.1', r));
  return { port: srv.address().port, observations, close: () => { srv.closeAllConnections(); srv.close(); } };
};
