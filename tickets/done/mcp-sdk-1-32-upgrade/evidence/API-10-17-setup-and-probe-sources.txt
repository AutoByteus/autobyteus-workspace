      cell.turnCompleted = msgs.some((m) => m.type === 'TURN_COMPLETED');

      const toolEvents = msgs.filter((m) => /^TOOL_EXECUTION_(SUCCEEDED|FAILED)$/.test(m.type)).map((m) => ({ type: m.type, tool: norm(m.payload.tool_name) }));
      cell.toolEvents = toolEvents;
      const ok = (t) => toolEvents.some((e) => e.tool === t && e.type === 'TOOL_EXECUTION_SUCCEEDED');
      const wire = JSON.stringify(msgs);
      const tasks = (await gql('query($p: String!) { projectTasks(projectId: $p) { taskKey description status } }', { p: project.projectId })).data?.projectTasks ?? [];

      cell.externalStdio = { succeeded: ok('video_editing_get_media_duration'), durationOnWire: Boolean(result.gateway.stdioTool?.text) && wire.includes(result.gateway.stdioTool.text) };
      cell.externalHttp = { succeeded: ok('parity_http_auth_probe'), markerReceived: wire.includes(HTTP_MARKER), remoteToolCalls: httpCalls.map((o) => o.auth) };
      cell.dummyOnWire = wire.includes(DUMMY);
      cell.columns = {

} finally {

          if (Date.now() > deadline) throw error;
          await new Promise((r) => setTimeout(r, 500));
        }
      }
    };
    const marker = `Parity codex direct ${randomUUID().slice(0, 8)}`;
    const created = await call("create_task", { project_id: project.projectId, description: marker });
    const createdResult = (JSON.parse(text(created)) as { task: { taskId: string; status: string } }).task;
    const updated = await call("update_task", { task_id: createdResult.taskId, status: "DONE" });
    const updatedResult = (JSON.parse(text(updated)) as { task?: { taskId: string; status: string } }).task ?? { taskId: "", status: text(updated).slice(0, 200) };
    const duration = await call("video_editing_get_media_duration", { media_path: sample });
    const httpMark = httpServer.toolCalls().length;
    const probe = await call("parity_http_auth_probe", {});
    const tasks = (await gql<{ projectTasks: Array<{ taskKey: string; description: string; status: string }> }>(
      "query($p: String!) { projectTasks(projectId: $p) { taskKey description status } }", { p: project.projectId })).projectTasks;
    Object.assign(evidence, {
      createTask: { isError: created.isError ?? false, status: createdResult.status },
      updateTask: { isError: updated.isError ?? false, status: updatedResult.status },
      savedTask: tasks.find((t) => t.description === marker)?.status ?? null,
      externalStdio: { isError: duration.isError ?? false, text: text(duration) },
      externalHttp: { isError: probe.isError ?? false, text: text(probe), remoteToolCalls: httpServer.toolCalls().slice(httpMark).map((o) => o.auth) },
    });
    if (process.env["PARITY_EVIDENCE"]) await fs.writeFile(process.env["PARITY_EVIDENCE"], JSON.stringify(evidence, null, 2));
    await gql("mutation($id: String!) { terminateAgentRun(agentRunId: $id) { success } }", { id: run.runId }).catch(() => undefined);
    expect(evidence).toMatchObject({ createTask: { isError: false, status: "TODO" }, updateTask: { isError: false, status: "DONE" }, savedTask: "DONE",
      externalStdio: { isError: false }, externalHttp: { isError: false, text: "MARKER-PARITY-CODEX-HTTP", remoteToolCalls: ["expected"] } });
    expect(Number((evidence.externalStdio as { text: string }).text)).toBeGreaterThan(0);
  }, 300_000);
});
