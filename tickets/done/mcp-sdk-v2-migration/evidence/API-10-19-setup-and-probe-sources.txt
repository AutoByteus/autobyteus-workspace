// Temporary API/E2E probe (ticket mcp-sdk-v2-migration, case API-14, RR-001).
// SDK-independent: speaks raw JSON-RPC over stdio to each real server, so neither SDK's validation can hide anything.
// Records per server: the protocol version it answers for a 2025-11-25 initialize, its tool count (all pages),
// and per tool whether it declares an outputSchema and which `$schema` dialect that schema names.
// Usage: node raw-dialect-probe.mjs <servers.json> <out.json>
import { spawn } from 'node:child_process';
import fs from 'node:fs';

const [serversFile, outFile] = process.argv.slice(2);
const servers = JSON.parse(fs.readFileSync(serversFile, 'utf8'));
const out = {};

const probe = (id, cfg) => new Promise((resolve) => {
  const child = spawn(cfg.command, cfg.args ?? [], { env: { ...process.env, ...(cfg.env ?? {}) }, cwd: cfg.cwd, stdio: ['pipe', 'pipe', 'ignore'] });
  let buf = '';
  const pending = new Map();
  let nextId = 1;
  const timer = setTimeout(() => { child.kill('SIGKILL'); resolve({ error: 'timeout after 120 s' }); }, 120_000);
  child.stdout.on('data', (d) => {
    buf += d;
    let i;
    while ((i = buf.indexOf('\n')) >= 0) {
      const line = buf.slice(0, i).trim(); buf = buf.slice(i + 1);
      if (!line) continue;
      try { const m = JSON.parse(line); if (m.id !== undefined && pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); } } catch { /* non-JSON line */ }
    }
  });
  child.on('exit', (code) => { clearTimeout(timer); for (const r of pending.values()) r({ error: { message: `exited ${code}` } }); });
  const rpc = (method, params) => new Promise((r) => { const id = nextId++; pending.set(id, r); child.stdin.write(JSON.stringify({ jsonrpc: '2.0', id, method, params }) + '\n'); });
  (async () => {
    const init = await rpc('initialize', { protocolVersion: '2025-11-25', capabilities: {}, clientInfo: { name: 'raw-dialect-probe', version: '1' } });
    if (init.error) { clearTimeout(timer); child.kill(); resolve({ error: init.error }); return; }
    child.stdin.write(JSON.stringify({ jsonrpc: '2.0', method: 'notifications/initialized' }) + '\n');
    const tools = []; let cursor; let pages = 0;
    do { const r = await rpc('tools/list', cursor ? { cursor } : {}); pages++; tools.push(...(r.result?.tools ?? [])); cursor = r.result?.nextCursor; } while (cursor && pages < 64);
    clearTimeout(timer); child.kill();
    const dialects = {};
    for (const t of tools) { const d = t.outputSchema ? (t.outputSchema.$schema ?? '(no $schema)') : '(no outputSchema)'; dialects[d] = (dialects[d] ?? 0) + 1; }
    const inputDialects = {};
    for (const t of tools) { const d = t.inputSchema?.$schema ?? '(no $schema)'; inputDialects[d] = (inputDialects[d] ?? 0) + 1; }
    resolve({ serverInfo: init.result?.serverInfo, protocolVersion: init.result?.protocolVersion, toolCount: tools.length, pages,
      outputSchemaDialects: dialects, inputSchemaDialects: inputDialects,
      toolsWithOutputSchema: tools.filter((t) => t.outputSchema).map((t) => t.name) });
  })();
});

for (const [id, cfg] of Object.entries(servers)) {
  out[id] = await probe(id, cfg);
  console.log(id, JSON.stringify(out[id]));
}
fs.writeFileSync(outFile, JSON.stringify(out, null, 2));
