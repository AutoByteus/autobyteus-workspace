# Test-only fake node for the autobyteus-433 SR-004 rendered check. Mode file: ok | 502 | drop.
import sys
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
MODE = sys.argv[2]
class H(BaseHTTPRequestHandler):
    def do_GET(self):
        mode = open(MODE).read().strip()
        if self.path.startswith('/rest/remote-access/status'):
            body = b'{"phoneAccessEnabled":true,"pairingAvailable":true}'; ctype = 'application/json'; code = 200
        elif mode == 'drop':
            self.connection.close(); return
        elif mode == '502':
            body = b'<html><body><h1>bad gateway</h1></body></html>'; ctype = 'text/html'; code = 502
        else:
            body = b'<html><head><meta name="viewport" content="width=device-width"></head><body style="background:#dbeafe"><h1>TEST PAGE OK</h1></body></html>'; ctype = 'text/html'; code = 200
        self.send_response(code); self.send_header('Content-Type', ctype); self.send_header('Content-Length', str(len(body))); self.send_header('Cache-Control', 'no-store'); self.end_headers(); self.wfile.write(body)
    def log_message(self, fmt, *a): sys.stderr.write((fmt % a) + '\n')
ThreadingHTTPServer(('127.0.0.1', int(sys.argv[1])), H).serve_forever()
