// Temporary API/E2E probe (autobyteus-224, API-03/API-04): MCP through a running AutoByteus server's public
// GraphQL and /mcp/gateway. Preview, configure + discover a stdio server (the repo's Node fixture, env and cwd set)
// and a streamable HTTP server (the same fixture in HTTP mode with a dummy token), then call tools via the gateway.
// Usage: node mcp-journey.mjs <backendUrl> <stdioCommand> <fixturePathAsSeenByServer> <httpUrlAsSeenByServer> <out.json> [gatewayToken]
import fs from 'node:fs';
import { Client, StreamableHTTPClientTransport } from '@modelcontextprotocol/client';
const [BACKEND, CMD, FIXTURE, HTTP_URL, OUT, GW_TOKEN] = process.argv.slice(2);
const gql = async (query, variables) => (await (await fetch(`${BACKEND}/graphql`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ query, variables }) })).json());
const timed = async (fn) => { const t0 = Date.now(); const v = await fn(); return { ms: Date.now() - t0, v }; };
const r = { backend: BACKEND, at: new Date().toISOString() };
const servers = [
  { serverId: 'hyg_stdio', transportType: 'STDIO', enabled: true, toolNamePrefix: 'hyg_stdio', stdioConfig: { command: CMD, args: [FIXTURE], env: { HYG_ENV: 'hyg-env-dummy' }, cwd: process.env.H_CWD ?? '/tmp', } },
  { serverId: 'hyg_http', transportType: 'STREAMABLE_HTTP', enabled: true, toolNamePrefix: 'hyg_http', streamableHttpConfig: { url: HTTP_URL, token: 'test-dummy', headers: null } },
];
r.health = (await fetch(`${BACKEND}/rest/health`)).status;
for (const input of servers) {
  const p = await timed(() => gql('query($input: McpServerInput!) { previewMcpServerTools(input: $input) { name } }', { input }));
  const c = await gql('mutation($input: McpServerInput!) { configureMcpServer(input: $input) { savedConfig { __typename } } }', { input });
  const d = await timed(() => gql('mutation($id: String!) { discoverAndRegisterMcpServerTools(serverId: $id) { success message discoveredTools { name } } }', { id: input.serverId }));
  r[input.serverId] = { preview: { ms: p.ms, tools: p.v.data?.previewMcpServerTools?.map((t) => t.name) ?? null, error: p.v.errors?.[0]?.message ?? null },
    configured: Boolean(c.data?.configureMcpServer), discover: { ms: d.ms, success: d.v.data?.discoverAndRegisterMcpServerTools?.success ?? null, message: d.v.data?.discoverAndRegisterMcpServerTools?.message ?? JSON.stringify(d.v.errors) } };
}
const client = new Client({ name: 'hyg-gateway-probe', version: '1' });
await client.connect(new StreamableHTTPClientTransport(new URL(`${BACKEND}/mcp/gateway`), { requestInit: { headers: GW_TOKEN ? { authorization: `Bearer ${GW_TOKEN}` } : {} } }));
const tools = (await client.listTools()).tools.map((t) => t.name);
r.gateway = { toolCount: tools.length, hygTools: tools.filter((t) => t.startsWith('hyg_')).sort(), calls: {} };
for (const [name, args] of [['hyg_stdio_report_env_and_cwd', { name: 'HYG_ENV' }], ['hyg_stdio_add', { a: 2, b: 3 }], ['hyg_stdio_fail', { message: 'hyg-tool-error' }], ['hyg_http_echo', { text: 'hyg-http-echo' }]]) {
  const c = await timed(() => client.callTool({ name, arguments: args }));
  r.gateway.calls[name] = { ms: c.ms, isError: c.v.isError ?? false, text: (c.v.content ?? []).map((x) => x.text).join(''), structured: c.v.structuredContent ?? null };
}
await client.close();
fs.writeFileSync(OUT, JSON.stringify(r, null, 2)); console.log(JSON.stringify(r, null, 1)); process.exit(0);
