/** All descendant processes of this test process whose command line looks like a runtime CLI. */
const runtimeProcesses = (needle: RegExp): Array<{ pid: number; command: string }> => {
  const rows = execFileSync("ps", ["-A", "-o", "pid=,ppid=,command="], { encoding: "utf-8" }).split("\n")
    .map((line) => line.trim().match(/^(\d+)\s+(\d+)\s+(.*)$/)).filter(Boolean)
    .map((m) => ({ pid: Number(m![1]), ppid: Number(m![2]), command: m![3] }));
  const descendants = new Set<number>([process.pid]);
  let grew = true;
  while (grew) { grew = false; for (const row of rows) if (descendants.has(row.ppid) && !descendants.has(row.pid)) { descendants.add(row.pid); grew = true; } }
  return rows.filter((row) => row.pid !== process.pid && descendants.has(row.pid) && needle.test(row.command)).map(({ pid, command }) => ({ pid, command: command.slice(0, 160) }));
};

      const suffix = randomUUID().slice(0, 6);
      const word = `PING${suffix.toUpperCase()}`;
      const hostId = await createDef(`Addr Host ${suffix}`,
        "You are a precise tool operator. When the user tells you to message a collaborator, call send_message_to exactly once as told, then reply with one short sentence. "
        + "When another agent messages you with a word, call send_message_to exactly once with recipient_address set to the sender address given in the message header (not the sender id) and content 'ACK ' followed by that word. Then reply 'done'.");
      const leadId = await createDef(`Addr Lead ${suffix}`,
        "When an agent asks you to send it a word, call send_message_to exactly once with recipient_address set to the sender address given in the message header and content set to exactly that word. "
        + "When a message starts with ACK, do not call any tool; reply 'ok'.");
      const mateId = await createDef(`Addr Mate ${suffix}`, "Reply in one short sentence without tools.");
      const team = await gql<{ createAgentTeamDefinition: { id: string } }>("mutation($input: CreateAgentTeamDefinitionInput!) { createAgentTeamDefinition(input: $input) { id } }", { input: {
        name: `Addr Team ${suffix}`, description: "Address team.", instructions: "Work together.", coordinatorMemberName: "lead",
        nodes: [{ memberName: "lead", ref: leadId, refScope: "SHARED" }, { memberName: "mate", ref: mateId, refScope: "SHARED" }], handoffs: [] } });
      teamDefinitionIds.add(team.createAgentTeamDefinition.id);
      const hostRunId = await createRun(hostId, runtime.runtimeKind, model);
      const host = await openSocket(`/ws/agent/${hostRunId}`);
      const t = await sendTurn(host, `Use send_message_to to message the mentioned team at its address, asking it to send you the word ${word}. Then reply with one short sentence.`,
        [{ kind: "agent_team", definition_id: team.createAgentTeamDefinition.id }]);
      expect(t.ack.payload.accepted, JSON.stringify(t.ack.payload)).toBe(true);
      const teamEntry = (await view(hostRunId))!.execution_tree.collaborators[0];
      const lead = teamEntry.members.find((m: any) => m.address.endsWith("/lead"));
      const mate = teamEntry.members.find((m: any) => m.address.endsWith("/mate"));
      log(runtime.runtimeKind, "team", teamEntry.address, "lead", lead.address, lead.agentRunId);
      let recorded: unknown = null;
      const flow = await poll("host → lead → host → lead(ACK)", async () => {
        const msgs = (await view(hostRunId))?.communication_messages.messages ?? [];
        recorded = msgs.map((m) => `${m.senderAgentRunId}→${m.receiverAgentRunId}: ${m.content.slice(0, 60)}`);
        const ping = msgs.find((m) => m.senderAgentRunId === lead.agentRunId && m.receiverAgentRunId === hostRunId && m.content.includes(word));
        const ack = msgs.find((m) => m.senderAgentRunId === hostRunId && m.receiverAgentRunId === lead.agentRunId && /ACK/.test(m.content));
        return ping && ack ? msgs : null;
      }).catch((e) => { throw new Error(`${e instanceof Error ? e.message : e} Recorded: ${JSON.stringify(recorded)}`); });
      log(runtime.runtimeKind, "flow", JSON.stringify(flow.map((m) => `${m.senderAgentRunId}→${m.receiverAgentRunId}: ${m.content.slice(0, 60)}`)));
      const sends = host.messages.filter((m) => m.type === "TOOL_EXECUTION_SUCCEEDED" && String(m.payload.tool_name ?? "").endsWith("send_message_to"));
      const sendArgs = host.messages.filter((m) => String(m.payload.tool_name ?? "").endsWith("send_message_to") && m.payload.arguments).map((m) => JSON.stringify(m.payload.arguments));
      log(runtime.runtimeKind, "host send_message_to args", JSON.stringify(sendArgs), "succeeded", sends.length);
      expect(sendArgs.some((a) => a.includes(`"${lead.address}"`) && /ACK/.test(a)), `host replied by lead address ${lead.address}: ${JSON.stringify(sendArgs)}`).toBe(true);
      const header = `sender address: ${lead.address}`;
      expect(await poll("host input carries the lead's sender address", async () => grepMemory(hostRunId, header), 30_000)).toBe(true);
      expect(await grepMemory(lead.agentRunId, `sender address: ${(await view(hostRunId))!.execution_tree.host.address}`), "lead input carries the host's address").toBe(true);
      await poll("lead conversation shows the ACK from the host", async () => (await memberConversation(hostRunId, lead.address, lead.agentRunId)).some((e) => e.kind === "inter_agent_message" && e.senderAgentRunId === hostRunId && /ACK/.test(e.content ?? "")));
      await wait(10_000);
      const usage = await Promise.all([lead.agentRunId, mate.agentRunId].map(exactSummary));
      log(runtime.runtimeKind, "member usage", JSON.stringify(usage.map((u) => [u.runId, u.totalTokens])));
      const children = [lead.agentRunId, ...(usage[1].totalTokens > 0 ? [mate.agentRunId] : [])];
      const rolled = await standaloneSummary(hostRunId);
      const host1 = await exactSummary(hostRunId);
      const expected = host1.totalTokens + usage.reduce((t, u) => t + u.totalTokens, 0);
      log(runtime.runtimeKind, "team roll-up", rolled.totalTokens, "expected", expected, "children", JSON.stringify(children));
      expect(rolled.totalTokens).toBe(expected);
      expect(rolled.latestPromptTokens).toBe(host1.latestPromptTokens);
    }, 2_400_000);
  }
});
