import { writeFileSync } from 'node:fs'
import { describe, expect, it, vi } from 'vitest'

// autobyteus-461 SR-069 (CR-461-035): base 827f1e68dc characterization. A run on another node takes no `@`: the
// composer's mention scope is null for it (autobyteus-278 REQ-808), so the web never sends mentions for a peer row.
const state = vi.hoisted(() => ({ remote: false }))
vi.mock('~/stores/activeContextStore', () => ({ useActiveContextStore: () => ({
  // A delegated row in an agent run's view (a peer's, when remoteRuns knows it).
  activeWorkspaceTarget: { kind: 'agent_run_task_agent', access: 'live', host: { hostRunId: 'host-run' }, context: { state: { runId: 'peer-agent-run' } } },
  send: vi.fn(), interruptGeneration: vi.fn(),
}) }))
vi.mock('~/stores/remoteRunsStore', () => ({ useRemoteRunsStore: () => ({ runOfAgentRun: () => (state.remote ? { nodeId: 'node-b' } : null) }) }))
vi.mock('~/composables/agentInput/runMentionScope', () => ({ resolveRunMentionScope: () => ({ kind: 'local-scope' }) }))

import { useComposerTarget } from '../useComposerTarget'

describe('base 827f1e68dc: the composer of a run on another node', () => {
  it('has no mention scope; a run on this node has one', () => {
    state.remote = true
    const peer = useComposerTarget().value?.mentionScope ?? null
    state.remote = false
    const local = useComposerTarget().value?.mentionScope ?? null
    writeFileSync(process.env.CHAR_OUT ?? '/tmp/char-461-peer-mentions.json', JSON.stringify({ peer, local }, null, 2))
    expect(peer).toBeNull()
  })
})
