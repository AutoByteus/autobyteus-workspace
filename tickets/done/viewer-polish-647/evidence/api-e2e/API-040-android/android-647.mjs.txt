// localabstract:webview_devtools_remote_<pid>). Usage: node android-647.mjs <cdpPort> <outDir>
// Steps: instrumentation registered for new documents, the page reloaded (the session starts there); pair if asked;
// AC-005: the run's Files → pdf-647 → alpha, beta, gamma, Back between, two rounds; worker and blob counts.
// AC-001: sample-647.json in the phone viewer: line numbers, margin, first character. AC-002: the Team's Messages →

import fs from 'node:fs/promises';
import path from 'node:path';
const [port, out, step = 'all'] = process.argv.slice(2);
const sleep = ms => new Promise(r => setTimeout(r, ms));
const targets = await (await fetch(`http://127.0.0.1:${port}/json`)).json();
const page = targets.find(t => t.type === 'page' && /\/mobile/.test(t.url)) ?? targets.find(t => t.type === 'page');
const ws = new WebSocket(page.webSocketDebuggerUrl);
await new Promise((resolve, reject) => { ws.onopen = resolve; ws.onerror = reject; });
let id = 0; const pending = new Map();
ws.onmessage = (m) => { const msg = JSON.parse(m.data); if (msg.id && pending.has(msg.id)) { pending.get(msg.id)(msg); pending.delete(msg.id); } };
const send = (method, params = {}) => new Promise((resolve) => { const i = ++id; pending.set(i, resolve); ws.send(JSON.stringify({ id: i, method, params })); });
const evaluate = async (expression) => {
  const r = await send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true, userGesture: true });
  if (r.result?.exceptionDetails) throw new Error(JSON.stringify(r.result.exceptionDetails).slice(0, 300));
  return r.result?.result?.value;
};
const waitFor = async (label, expression, ms = 30000) => {
  const end = Date.now() + ms;
  while (Date.now() < end) { if (await evaluate(expression).catch(() => false)) return true; await sleep(400); }
  throw new Error(`timeout: ${label}`);
};
const click = sel => evaluate(`(() => { const e = ${sel}; if (!e) return false; e.click(); return true; })()`);
const result = { steps: [] };
const log = (s, d) => { result.steps.push({ step: s, ...d }); console.log(s, JSON.stringify(d).slice(0, 500)); };
const instrument = `(() => { try { if (window.__vp647) return; const rec = { blobs: [], workers: [], workerErrors: 0 };
  const c = URL.createObjectURL.bind(URL); URL.createObjectURL = (o) => { const u = c(o); rec.blobs.push({ type: o && o.type !== undefined ? o.type : typeof o }); return u; };
  const W = window.Worker; if (W) { const P = function (url, opts) { const w = new W(url, opts); rec.workers.push({ scheme: String(url).split(':')[0], type: (opts && opts.type) || 'classic' }); w.addEventListener('error', () => { rec.workerErrors += 1; }); return w; }; P.prototype = W.prototype; window.Worker = P; }
  window.__vp647 = rec; } catch (e) {} })();`;

await send('Page.enable');
await send('Runtime.enable');

  const runOpen = `[...document.querySelectorAll('[data-testid^="mobile-run-row-agent:"]')].find(r => /Video Checker/.test(r.textContent))?.querySelector('[data-testid="mobile-run-open"]')`;
  await waitFor('the run row', `Boolean(${runOpen})`, 30000);
  await click(runOpen);
  await waitFor('the run screen', `Boolean(document.querySelector('[data-testid="mobile-run-menu-button"]'))`);
  await sleep(800);
  await click(`document.querySelector('[data-testid="mobile-run-menu-button"]')`);
  await waitFor('the menu', `Boolean(document.querySelector('[data-testid="mobile-run-menu-files"]'))`);
  await click(`document.querySelector('[data-testid="mobile-run-menu-files"]')`);
  await waitFor('the files list', `Boolean(document.querySelector('[data-testid="mobile-files-list"]'))`);
  const row = name => `[...document.querySelectorAll('[data-testid="mobile-files-row"]')].find(r => r.textContent.includes(${JSON.stringify(name)}))`;
  await waitFor('pdf-647', `Boolean(${row('pdf-647')})`);
  await click(row('pdf-647'));
  await waitFor('in pdf-647', `(document.querySelector('[data-testid="mobile-files-path"]')?.textContent ?? '').includes('pdf-647')`);

  const back = async () => {

      await click(`[...document.querySelectorAll('[data-testid="mobile-screen-back"]')].filter(b => b.offsetParent).pop()`);

[])].filter(x => x.width > 0 && x.getBoundingClientRect().width > 0); const a = v?.querySelector('[role="alert"]'); const inked = c.filter(x => { try { const d = x.getContext('2d').getImageData(0, 0, x.width, Math.min(x.height, 400)).data; for (let i = 0; i < d.length; i += 4 * 97) { if (d[i + 3] > 0 && (d[i] < 200 || d[i + 1] < 200 || d[i + 2] < 200)) return true; } return false; } catch (e) { return false; } }).length; return { pages: c.length, inked, error: a ? a.textContent.trim().slice(0, 160) : null }; })()`;
      await waitFor(`${name} drawn`, `(${state}).pages >= ${pages} || Boolean((${state}).error)`, 30000).catch(() => undefined);
