--- inspect.cjs ---
const {chromium}=require('../node_modules/playwright-core');
const fs=require('node:fs');
(async()=>{
 const browser=await chromium.launch({executablePath:'/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',headless:true});
 const page=await browser.newPage();
 const out='../tickets/in-progress/task-page-copy-simplification/evidence';
 const notes=[];
 page.on('pageerror',e=>notes.push({error:e.message}));
 for(const mode of ['new','edit']){
  for(const [name,width,height] of [['wide',1280,900],['narrow',390,844]]){
   await page.setViewportSize({width,height});
   await page.goto('http://127.0.0.1:4197?mode='+mode);
   await page.locator('textarea').waitFor();
   await page.screenshot({path:out+'/'+mode+'-'+name+'.png',fullPage:true,timeout:15000});
   notes.push({mode,viewport:name,text:await page.locator('body').innerText(),layout:await page.evaluate(()=>({scrollWidth:document.documentElement.scrollWidth,width:innerWidth,label:document.querySelector('label').getBoundingClientRect().y,textarea:document.querySelector('textarea').getBoundingClientRect().y,heading:document.activeElement?.tagName}))});
   await page.locator('textarea').fill('   ');
   await page.locator('[data-testid=task-page-save]').click();
   notes.push({mode,viewport:name,blank:await page.evaluate(()=>({focused:document.activeElement?.id,error:document.querySelector('#task-page-error')?.textContent,describedBy:document.querySelector('textarea').getAttribute('aria-describedby')}))});
   await page.screenshot({path:out+'/'+mode+'-'+name+'-blank.png',fullPage:true,timeout:15000});
   await page.locator('textarea').fill('Corrected description');
   notes.push({mode,viewport:name,corrected:await page.evaluate(()=>({text:document.querySelector('textarea').value,error:!!document.querySelector('#task-page-error'),describedBy:document.querySelector('textarea').getAttribute('aria-describedby')}))});
  }
 }
 await page.setViewportSize({width:390,height:844});
 await page.goto('http://127.0.0.1:4197?locale=zh-CN');
 await page.locator('textarea').waitFor();
 notes.push({locale:'zh-CN',text:await page.locator('body').innerText(),placeholder:await page.locator('textarea').getAttribute('placeholder')});
 await page.screenshot({path:out+'/new-zh-narrow.png',fullPage:true,timeout:15000});
 fs.writeFileSync(out+'/render-inspection.json',JSON.stringify(notes,null,2));
 await browser.close();
})();
