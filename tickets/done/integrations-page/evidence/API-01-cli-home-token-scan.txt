```js
// Temporary token scan (autobyteus-337 API/E2E, AC-007): reads test.github.token from the test vault and reports,
// for each given file or folder, the files whose bytes contain it, and the files containing an optional marker
// (to show the scanned files really hold the runs' output). Prints paths and counts only, never the token.
import fs from "node:fs";
import path from "node:path";
const WT = "/Users/normy/autobyteus_org/autobyteus-worktrees/integrations-page";
const { openVerifiedTestVault } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/open-test-vault.js`);
const { readSecretEntry } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/read-secret-entry.js`);
const { secretId } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/secret-id.js`);
const vault = await openVerifiedTestVault(process.env.HOME + "/.autobyteus-test/vault");
const token = (await readSecretEntry(vault.store, vault.rootKey, vault.metadata, secretId("test.github.token"))).revealToTrustedConsumer();
vault.close();
const marker = process.env.SCAN_MARKER || null;
const files = (p) => { if (!fs.existsSync(p)) return []; const st = fs.lstatSync(p); if (st.isFile()) return [p];
  if (!st.isDirectory()) return []; return fs.readdirSync(p).flatMap((n) => files(path.join(p, n))); };
const report = {};
for (const target of process.argv.slice(2)) {
  const list = files(target);
  const withToken = [], withMarker = [];
  for (const f of list) { const b = fs.readFileSync(f); if (b.includes(token)) withToken.push(f); if (marker && b.includes(marker)) withMarker.push(f); }
  report[target] = { files: list.length, filesWithToken: withToken, ...(marker ? { filesWithMarker: withMarker.length } : {}) };
}
console.log(JSON.stringify(report, null, 2));
```
