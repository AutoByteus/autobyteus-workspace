// Shared helpers for the temporary desktop drivers (autobyteus-337 API/E2E, API-04/API-05). Never prints the token.
import fs from "node:fs";
import path from "node:path";
export const WT = "/Users/normy/autobyteus_org/autobyteus-worktrees/integrations-page";
export const mask = (t) => String(t).replace(/rp_[A-Za-z0-9_-]+/g, "rp_<pass>");
export const log = (step, data = {}) => console.log(mask(JSON.stringify({ step, ...data })));
export const wait = (ms) => new Promise((r) => setTimeout(r, ms));
export const readToken = async () => {
  const { openVerifiedTestVault } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/open-test-vault.js`);
  const { readSecretEntry } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/read-secret-entry.js`);
  const { secretId } = await import(`${WT}/autobyteus-ts/dist/secrets/vault/secret-id.js`);
  const vault = await openVerifiedTestVault(`${process.env.HOME}/.autobyteus-test/vault`);
  try { return (await readSecretEntry(vault.store, vault.rootKey, vault.metadata, secretId("test.github.token"))).revealToTrustedConsumer(); }
  finally { vault.close(); }
};
export const connectApp = async (controlPort) => {
  const { chromium } = await import(`${WT}/autobyteus-web/node_modules/playwright-core/index.mjs`);
  const browser = await chromium.connectOverCDP(`http://127.0.0.1:${controlPort}`);
  const page = browser.contexts()[0].pages().find((p) => !p.url().startsWith("devtools")) ?? browser.contexts()[0].pages()[0];
  page.on("dialog", (dialog) => { log("page-dialog", { message: dialog.message() }); dialog.accept(); });
  await page.setViewportSize({ width: 1440, height: 900 }).catch(() => undefined);
  return page;
};
export const gqlFor = (serverPort) => async (query, variables) => {
  const res = await fetch(`http://127.0.0.1:${serverPort}/graphql`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ query, variables }) });
  const body = await res.json();
  if (!body.data || body.errors?.length) throw new Error(JSON.stringify(body.errors ?? body));
  return body.data;
};
/** Sets the hash until `selector` shows: the renderer may settle on its start route after a load. */
export const go = async (page, hash, selector) => {
  const target = page.locator(selector).first();
  for (let i = 0; i < 10 && !(await target.isVisible().catch(() => false)); i++) {
    await page.evaluate((h) => { location.hash = h; }, hash);
    await target.waitFor({ timeout: 5_000 }).catch(() => undefined);
  }
  await target.waitFor({ timeout: 5_000 });
};
export const shotter = (page, dir) => async (name) => { fs.mkdirSync(dir, { recursive: true }); await page.screenshot({ path: path.join(dir, name) }); log("screenshot", { file: name }); };
export const connectionsFile = (dataRoot) => {
  const f = path.join(dataRoot, "server-data", "integrations", "connections.json");
  return fs.existsSync(f) ? JSON.parse(fs.readFileSync(f, "utf8")) : null;
};

// G. A real Codex agent uses the integration's scripts (AC-008).
const stamp = `AutoByteus autobyteus-337 API/E2E ${new Date().toISOString()}`;
const workspace = path.join(dataRoot, "agent-workspace"); fs.mkdirSync(workspace, { recursive: true });
let runId = process.env.RUN_ID ?? null; let wire = [];
const turn = async (content, timeoutMs = 600_000) => {
  const ws = new WebSocket(`ws://127.0.0.1:${serverPort}/ws/agent/${runId}`);
  const got = [];
  ws.addEventListener("message", (e) => { try { const m = JSON.parse(String(e.data)); if (m.type) got.push(m); } catch {} });
  await new Promise((resolve, reject) => { ws.addEventListener("open", resolve); ws.addEventListener("error", reject); });
  const id = `api-e2e-${Date.now()}`;
  ws.send(JSON.stringify({ type: "SEND_MESSAGE", payload: { agent_run_id: runId, content, message_id: id, dedupe_key: `agent_run_input:e2e:${id}`, context_file_paths: [], image_urls: [] } }));
  const ends = new Set(["TURN_COMPLETED", "ERROR", "TOOL_APPROVAL_REQUESTED"]);
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline && !got.some((m) => ends.has(m.type))) await wait(250);
  ws.close(); wire = wire.concat(got);
  return { endTypes: got.map((m) => m.type).filter((t) => ends.has(t)), messages: got.length };
};
