===== lib.mjs
// Temporary API/E2E desktop driver helpers (ticket autobyteus-244). Attaches to an isolated AutoByteus instance's
// own window over CDP (never launches or closes the app); GraphQL and /mcp/gateway on its backend; reads its files.
import fs from 'node:fs';
import path from 'node:path';
import { createRequire } from 'node:module';
export const W = '/Users/normy/autobyteus_org/autobyteus-worktrees/secrets-and-integrations-design';
const req = createRequire(`${W}/autobyteus-web/package.json`);
const { chromium } = req('playwright-core');
const SDK_CLIENT = `${W}/autobyteus-ts/node_modules/@modelcontextprotocol/client/dist/index.mjs`; // MCP SDK v2
export const FIXTURE = `${W}/autobyteus-ts/tests/fixtures/mcp-stdio-fixture-server.mjs`;
export const NODE = process.execPath;

export const instance = (startJson) => JSON.parse(fs.readFileSync(startJson, 'utf8')).result;
export const attach = async (inst) => {
  const browser = await chromium.connectOverCDP(`http://127.0.0.1:${inst.controlPort}`);
  const page = browser.contexts()[0].pages().find((p) => p.url().startsWith('file:'));
  const errors = []; page.on('pageerror', (e) => errors.push(String(e)));
  return { browser, page, errors };
};
export const wait = (ms) => new Promise((r) => setTimeout(r, ms));
export const gql = async (inst, query, variables = {}) => (await (await fetch(`${inst.backendUrl}/graphql`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ query, variables }) })).json());
export const SERVERS_Q = `{ mcpServers { ... on StdioMcpServerConfig { serverId transportType enabled toolNamePrefix command args env { name value state } }
  ... on StreamableHttpMcpServerConfig { serverId transportType enabled toolNamePrefix url tokenState headers { name value state } } } }`;
export const servers = async (inst) => (await gql(inst, SERVERS_Q)).data?.mcpServers ?? [];
export const mcpsJson = (inst) => JSON.parse(fs.readFileSync(path.join(inst.dataRoot, 'server-data', 'mcps.json'), 'utf8'));
export const gatewayCall = async (inst, name, args = {}) => {
  const { Client, StreamableHTTPClientTransport } = await import(SDK_CLIENT);
  const c = new Client({ name: 's244-gateway-probe', version: '1' });
  try {
    await c.connect(new StreamableHTTPClientTransport(new URL(`${inst.backendUrl}/mcp/gateway`)));
    const listed = (await c.listTools()).tools.map((t) => t.name);
    if (!listed.includes(name)) return { listed: false };
    const r = await c.callTool({ name, arguments: args });
    return { listed: true, isError: r.isError ?? false, text: (r.content ?? []).map((x) => x.text ?? '').join('') };
  } catch (e) { return { error: String(e) }; } finally { await c.close().catch(() => {}); }
};
export const openMcp = async (page) => {
  await page.evaluate(() => { location.hash = '#/settings'; }); await wait(1200);
  await page.getByRole('button', { name: 'MCP Servers', exact: true }).click(); await wait(800);
  // Leave a sub-view (form or bulk import) the panel may remember.
  for (let i = 0; i < 4; i++) {
    if (await page.getByRole('button', { name: 'Add Server', exact: true }).count()) break;
    if (await page.locator('#serverId').count()) { await page.getByRole('button', { name: 'Cancel', exact: true }).first().click(); await wait(600); continue; }
    if (await page.getByRole('button', { name: 'Back', exact: true }).count()) { await page.getByRole('button', { name: 'Back', exact: true }).first().click(); await wait(600); }
  }
};
export const card = (page, id) => page.locator('div.rounded-lg.shadow-sm', { has: page.locator('h3', { hasText: new RegExp(`^${id}$`) }) });
export const shot = async (page, out, name) => { await page.screenshot({ path: path.join(out, `${name}.png`) }); };
export const bodyText = (page) => page.evaluate(() => document.body.innerText);

===== journey-a.mjs
// Temporary API/E2E journey A (autobyteus-244, API-03): stdio secret lifecycle and HTTP token states in the packaged
// app's own window. Usage: node journey-a.mjs <start.json> <outDir> <phase: stdio|http|after-restart> [httpUrl]
import fs from 'node:fs';
import path from 'node:path';
import { instance, attach, wait, servers, mcpsJson, gatewayCall, openMcp, card, shot, bodyText, NODE, FIXTURE } from './lib.mjs';

const [startJson, out, phase, httpUrl] = process.argv.slice(2);
const inst = instance(startJson);
const { browser, page, errors } = await attach(inst);
const facts = { phase };
const STDIO_TOKEN = 'test-dummy-github-token-0001';
const toastLines = async (before) => (await bodyText(page)).split('\n').map((s) => s.trim()).filter((s) => s && !before.includes(s)).slice(0, 6);
const rowFacts = (kind) => page.evaluate((k) => [...document.querySelectorAll(`[data-testid="mcp-row-${k}"]`)].map((row) => ({
  name: row.querySelector('[data-testid="mcp-row-name"]')?.value, nameReadonly: row.querySelector('[data-testid="mcp-row-name"]')?.readOnly,
  value: row.querySelector('[data-testid="mcp-row-value"]')?.value, type: row.querySelector('[data-testid="mcp-row-value"]')?.type,
  locked: row.querySelector('[data-testid="mcp-row-lock"]')?.getAttribute('aria-pressed') ?? null,
  move: Boolean(row.querySelector('[data-testid="mcp-row-move"]')), note: row.querySelector('[data-testid="mcp-row-note"]')?.innerText ?? null })), kind);
const editServer = async (id) => { await openMcp(page); await card(page, id).getByRole('button', { name: 'Edit', exact: true }).click(); await page.waitForSelector('#serverId'); await wait(500); };
const save = async () => { const before = await bodyText(page); await page.getByRole('button', { name: 'Save Configuration' }).click(); await wait(4000); return toastLines(before); };
const tab = async (name) => { await page.getByRole('button', { name, exact: true }).click(); await wait(400); };
const fileEntry = (id) => mcpsJson(inst).mcpServers[id];
const view = async (id) => (await servers(inst)).find((s) => s.serverId === id);

try {
  if (phase === 'stdio') {
    // Add through the JSON View, apply to the form: the rule pre-locks GITHUB_TOKEN; LOG_LEVEL stays a plain setting.
    await openMcp(page);
    await page.getByRole('button', { name: 'Add Server', exact: true }).click(); await page.waitForSelector('#serverId');
    await tab('JSON View');
    await page.fill('#json-input', JSON.stringify({ mcpServers: { 'github-like': { transportType: 'stdio', command: NODE, args: [FIXTURE], toolNamePrefix: 'gh', env: { GITHUB_TOKEN: STDIO_TOKEN, LOG_LEVEL: 'debug' } } } }, null, 2));
    await page.getByRole('button', { name: 'Apply JSON to Form' }).click(); await wait(800);
    facts.formRowsAfterApply = await rowFacts('env');
    if (!(await page.locator('#sync-on-save').isChecked())) await page.locator('#sync-on-save').check();
    await shot(page, out, 'API-03-01-stdio-form-prelocked');
    facts.saveToasts = await save();
  }
  if (phase === 'stdio' || phase === 'stdio-edit') {
    facts.view = await view('github-like');
    facts.fileEntry = fileEntry('github-like');
    facts.fileHasToken = JSON.stringify(mcpsJson(inst)).includes(STDIO_TOKEN);
    facts.gateway = await gatewayCall(inst, 'gh_report_env_and_cwd', { name: 'GITHUB_TOKEN' });
    facts.gatewayDelivered = facts.gateway.text ? JSON.parse(facts.gateway.text).value === STDIO_TOKEN : false;
    // Edit: the stored row is hidden, its name read-only; the JSON View shows a keep marker, never the value.
    await editServer('github-like');
    facts.editRows = await rowFacts('env');
    await shot(page, out, 'API-03-02-stdio-edit-stored-in-vault');
    await tab('JSON View');
    facts.jsonView = await page.inputValue('#json-input');
    facts.jsonViewHasToken = facts.jsonView.includes(STDIO_TOKEN);
    await shot(page, out, 'API-03-03-stdio-json-keep-marker');
    await page.getByRole('button', { name: 'Cancel', exact: true }).first().click();
  }
  if (phase === 'http') {
    const T_A = 'test-dummy-http-token-A', T_B = 'test-dummy-http-token-B';
    const step = async (label, fn) => { try { facts[label] = await fn(); } catch (e) { facts[label] = { error: String(e) }; } };
    await step('add-with-wrong-token', async () => {
      await openMcp(page);
      await page.getByRole('button', { name: 'Add Server', exact: true }).click(); await page.waitForSelector('#serverId');
      await page.fill('#serverId', 'remote-api'); await page.selectOption('#transportType', 'STREAMABLE_HTTP'); await wait(300);
      await page.fill('#toolNamePrefix', 'ra'); await page.fill('#http_url', httpUrl); await page.fill('#http_token', T_A);
      if (!(await page.locator('#sync-on-save').isChecked())) await page.locator('#sync-on-save').check();
      const toasts = await save();
      return { toasts, view: await view('remote-api'), fileToken: fileEntry('remote-api')?.token, gateway: await gatewayCall(inst, 'ra_echo', { text: 'hi' }) };
    });
    await step('replace-token', async () => {
      await editServer('remote-api');
      const note = await page.locator('[data-testid="mcp-token-note"]').innerText().catch(() => null);
      const placeholder = await page.getAttribute('#http_token', 'placeholder');
      const refBefore = fileEntry('remote-api')?.token?.vaultSecretId;
      await page.fill('#http_token', T_B);
      const toasts = await save();
      return { noteBefore: note, placeholder, toasts, view: await view('remote-api'), refChanged: fileEntry('remote-api')?.token?.vaultSecretId !== refBefore, gateway: await gatewayCall(inst, 'ra_echo', { text: 'hi' }) };
    });
    await step('keep-token', async () => {
      await editServer('remote-api');
      const refBefore = fileEntry('remote-api')?.token?.vaultSecretId;
      await page.fill('#toolNamePrefix', 'ra'); // unchanged save, token field left empty
      const toasts = await save();
      return { toasts, view: await view('remote-api'), refKept: fileEntry('remote-api')?.token?.vaultSecretId === refBefore, gateway: await gatewayCall(inst, 'ra_echo', { text: 'kept' }) };
    });
    await step('json-keep-marker', async () => {
      await editServer('remote-api'); await tab('JSON View');
      const text = await page.inputValue('#json-input');
      await shot(page, out, 'API-03-04-http-json-keep-marker');
      const before = await bodyText(page);
      await page.getByRole('button', { name: 'Save Configuration' }).click(); await wait(4000);
      return { jsonShowsMarker: text.includes('"stored": "vault"'), jsonHasToken: /test-dummy-http-token/.test(text), toasts: await toastLines(before), view: await view('remote-api'), gateway: await gatewayCall(inst, 'ra_echo', { text: 'json-kept' }) };
    });
    await step('invalid-keep-marker', async () => {
      await editServer('remote-api'); await tab('JSON View');
      const text = (await page.inputValue('#json-input')).replace('"stored": "vault"', '"stored": "bogus"');
      await page.fill('#json-input', text);
      const refBefore = fileEntry('remote-api')?.token?.vaultSecretId;
      const before = await bodyText(page);
      await page.getByRole('button', { name: 'Save Configuration' }).click(); await wait(3000);
      const lines = await toastLines(before);
      await shot(page, out, 'API-03-05-http-invalid-keep-marker');
      const r = { messages: lines, stillOnForm: Boolean(await page.locator('#serverId').count()), refUnchanged: fileEntry('remote-api')?.token?.vaultSecretId === refBefore, view: await view('remote-api') };
      if (r.stillOnForm) await page.getByRole('button', { name: 'Cancel', exact: true }).first().click();
      return r;
    });
    await step('remove-token', async () => {
      await editServer('remote-api');
      await page.locator('[data-testid="mcp-token-remove"]').check();
      const toasts = await save();
      return { toasts, view: await view('remote-api'), fileToken: fileEntry('remote-api')?.token ?? null, gateway: await gatewayCall(inst, 'ra_echo', { text: 'after-remove' }) };
    });
    facts.fileHasHttpToken = /test-dummy-http-token/.test(JSON.stringify(mcpsJson(inst)));
  }
  if (phase === 'after-restart') {
    let g = null;
    for (let i = 0; i < 40; i++) { g = await gatewayCall(inst, 'gh_report_env_and_cwd', { name: 'GITHUB_TOKEN' }); if (g.listed) break; await wait(500); }
    facts.gateway = g;
    facts.gatewayDelivered = g?.text ? JSON.parse(g.text).value === STDIO_TOKEN : false;
    facts.fileEntry = fileEntry('github-like');
    facts.fileHasToken = JSON.stringify(mcpsJson(inst)).includes(STDIO_TOKEN);
    facts.view = await view('github-like');
    await openMcp(page); await shot(page, out, 'API-03-06-list-after-restart');
  }
} catch (e) {
  facts.error = String(e);
} finally {
  facts.pageErrors = errors;
  fs.writeFileSync(path.join(out, `journey-a-${phase}.json`), JSON.stringify(facts, null, 2));
  console.log(JSON.stringify(facts, null, 2));
  await browser.close();
}

===== journey-b.mjs
// Temporary API/E2E journey B (autobyteus-244, API-04…API-07) on an isolated instance with an owned data root.
// Usage: node journey-b.mjs <start.json> <outDir> <phase> [extra]
//   phases: seed-check | move | keep | after-port-change | import <a-mcps.json> | replace | reimport <older-copy> |
//           vault-unavailable | vault-restored | zh
import fs from 'node:fs';
import path from 'node:path';
import { instance, attach, wait, gql, servers, mcpsJson, gatewayCall, openMcp, card, shot, bodyText } from './lib.mjs';

const [startJson, out, phase, extra] = process.argv.slice(2);
const inst = instance(startJson);
const { browser, page, errors } = await attach(inst);
const facts = { phase, backendUrl: inst.backendUrl };
const REVIEW_KEY = 'autobyteus.mcp.secretReview.embedded-local';
const view = async (id) => (await servers(inst)).find((s) => s.serverId === id);
const banner = () => page.locator('[data-testid="mcp-secret-move-banner"]');
const bannerFacts = async () => ({ visible: (await banner().count()) > 0, text: (await banner().count()) ? (await banner().innerText()).replace(/\s+/g, ' ') : null,
  reviewState: await page.evaluate((k) => localStorage.getItem(k), REVIEW_KEY) });
const toastLines = async (before) => (await bodyText(page)).split('\n').map((s) => s.trim()).filter((s) => s && !before.includes(s)).slice(0, 8);
const editServer = async (id) => { await openMcp(page); await card(page, id).getByRole('button', { name: 'Edit', exact: true }).click(); await page.waitForSelector('#serverId'); await wait(500); };
const rowFacts = (kind) => page.evaluate((k) => [...document.querySelectorAll(`[data-testid="mcp-row-${k}"]`)].map((row) => ({
  name: row.querySelector('[data-testid="mcp-row-name"]')?.value, value: row.querySelector('[data-testid="mcp-row-value"]')?.value,
  locked: row.querySelector('[data-testid="mcp-row-lock"]')?.getAttribute('aria-pressed') ?? null,
  move: Boolean(row.querySelector('[data-testid="mcp-row-move"]')), note: row.querySelector('[data-testid="mcp-row-note"]')?.innerText ?? null })), kind);
const waitTool = async (name, args) => { let g = null; for (let i = 0; i < 40; i++) { g = await gatewayCall(inst, name, args); if (g.listed) break; await wait(500); } return g; };
const fileMode = (p) => (fs.statSync(p).mode & 0o777).toString(8);

try {
  if (phase === 'seed-check') {
    // The seeded plain file loads unchanged; the banner lists its candidates.
    await openMcp(page); await wait(1000);
    facts.banner = await bannerFacts();
    facts.views = await servers(inst);
    facts.candidates = (await gql(inst, '{ mcpSecretMoveCandidates { serverId field matchedBy } }')).data?.mcpSecretMoveCandidates;
    facts.plainToolWorks = await waitTool('ia_report_env_and_cwd', { name: 'AUTOBYTEUS_API_KEY' });
    await shot(page, out, 'API-04-01-banner-on-seeded-plain-file');
  }
  if (phase === 'move') {
    // Review dialog: untick the false hit (a key-file path), move the rest. Only the move writes the 0600 backup.
    await openMcp(page); await wait(800);
    const before = await bodyText(page);
    await page.locator('[data-testid="mcp-secret-move-review"]').click(); await wait(800);
    const boxes = page.locator('[data-testid="mcp-secret-move-candidate"]');
    facts.dialogCandidates = await boxes.evaluateAll((els) => els.map((e) => ({ id: e.id, checked: e.checked })));
    const sshBox = page.locator('[data-testid="mcp-secret-move-candidate"][id*="SSH_MCP_PRIVATE_KEY_FILE"]');
    if (await sshBox.count()) await sshBox.uncheck();
    await shot(page, out, 'API-04-02-dialog-one-unticked');
    await page.locator('[data-testid="mcp-secret-move-confirm"]').click(); await wait(4000);
    facts.result = await page.locator('[data-testid="mcp-secret-move-result"]').innerText().catch(() => null);
    await shot(page, out, 'API-04-03-move-result');
    facts.toasts = await toastLines(before);
    const dir = path.join(inst.dataRoot, 'server-data');
    const backups = fs.readdirSync(dir).filter((f) => f.startsWith('mcps.json.backup-before-vault-move-'));
    facts.backups = backups.map((f) => ({ file: f, mode: fileMode(path.join(dir, f)), holdsPlainValues: fs.readFileSync(path.join(dir, f), 'utf8').includes('test-dummy-api-key-0002') }));
    facts.file = mcpsJson(inst).mcpServers;
    facts.fileHasMovedValues = /test-dummy-api-key-0002|test-dummy-http-token-B/.test(JSON.stringify(facts.file));
    facts.views = await servers(inst);
    facts.movedStdioWorks = await waitTool('ia_report_env_and_cwd', { name: 'AUTOBYTEUS_API_KEY' });
    facts.movedHttpWorks = await waitTool('lh_echo', { text: 'after-move' });
    const close = page.getByRole('button', { name: /Close|Done/ }).first(); if (await close.count()) { await close.click(); await wait(500); }
    await openMcp(page); await wait(800);
    facts.bannerAfter = await bannerFacts();
    await shot(page, out, 'API-04-04-list-after-move');
  }
  if (phase === 'keep') {
    // A new plain candidate appears; "Keep as they are" marks it reviewed.
    const exists = (await servers(inst)).some((s) => s.serverId === 'notes-server');
    const r = exists ? { errors: null } : await gql(inst, 'mutation($input: McpServerInput!) { configureMcpServer(input: $input) { savedConfig { __typename } } }', { input: {
      serverId: 'notes-server', transportType: 'STDIO', enabled: true, toolNamePrefix: 'ns',
      stdioConfig: { command: process.execPath, args: ['--version'], env: [{ name: 'NOTES_SERVICE_TOKEN', secret: false, value: 'test-dummy-notes-token' }], cwd: null } } });
    facts.configureErrors = r.errors ?? null;
    // The open page does not refetch after an out-of-band GraphQL change; a reload stands in for the UI's own refresh.
    await page.reload(); await wait(3000);
    await openMcp(page); await wait(800);
    facts.bannerBefore = await bannerFacts();
    await shot(page, out, 'API-04-05-banner-new-candidate');
    await page.locator('[data-testid="mcp-secret-move-keep"]').click(); await wait(800);
    facts.bannerAfter = await bannerFacts();
    await page.reload(); await wait(2500); await openMcp(page); await wait(800);
    facts.bannerAfterReload = await bannerFacts();
    facts.notesStillPlain = (await view('notes-server'))?.env;
  }
  if (phase === 'after-port-change') {
    await openMcp(page); await wait(1000);
    facts.banner = await bannerFacts();
    facts.candidatesFromServer = (await gql(inst, '{ mcpSecretMoveCandidates { serverId field matchedBy } }')).data?.mcpSecretMoveCandidates;
    facts.movedStdioWorks = await waitTool('ia_report_env_and_cwd', { name: 'AUTOBYTEUS_API_KEY' });
    await shot(page, out, 'API-04-06-list-after-restart-new-port');
  }
  if (phase === 'import') {
    // A file from another install (install A) holds references that mean nothing here.
    await openMcp(page);
    await page.getByRole('button', { name: 'Bulk Import', exact: true }).click(); await page.waitForSelector('#bulk-json-input');
    await page.fill('#bulk-json-input', fs.readFileSync(extra, 'utf8'));
    const before = await bodyText(page);
    await page.getByRole('button', { name: 'Import Configurations' }).click(); await wait(3000);
    facts.toasts = await toastLines(before);
    facts.stillOnView = Boolean(await page.locator('#bulk-json-input').count());
    facts.needsSecretList = await page.evaluate(() => [...document.querySelectorAll('div')].filter((d) => /secrets on another node/.test(d.innerText) && d.children.length < 6).map((d) => d.innerText).slice(-1)[0] ?? null);
    await shot(page, out, 'API-05-01-import-needs-secret');
    await openMcp(page); await wait(800);
    facts.card = await card(page, 'github-like').locator('[data-testid="mcp-card-needs-secret"]').innerText().catch(() => null);
    await shot(page, out, 'API-05-02-card-needs-secret');
    facts.view = await view('github-like');
    facts.fileEntry = mcpsJson(inst).mcpServers['github-like'];
    // Sync (discover) must not connect with an empty value.
    const d = await gql(inst, 'mutation($id: String!) { discoverAndRegisterMcpServerTools(serverId: $id) { success message } }', { id: 'github-like' });
    facts.discover = d.data?.discoverAndRegisterMcpServerTools ?? d.errors;
    fs.copyFileSync(path.join(inst.dataRoot, 'server-data', 'mcps.json'), path.join(out, '..', 'b-mcps-older-copy.json'));
  }
  if (phase === 'replace') {
    await editServer('github-like');
    facts.rowsBefore = await rowFacts('env');
    await shot(page, out, 'API-05-03-form-needs-secret-row');
    const row = page.locator('[data-testid="mcp-row-env"]', { has: page.locator('[data-testid="mcp-row-name"][value="GITHUB_TOKEN"]') });
    const valueInput = (await row.count()) ? row.locator('[data-testid="mcp-row-value"]') : page.locator('[data-testid="mcp-row-env"]').first().locator('[data-testid="mcp-row-value"]');
    await valueInput.fill('test-dummy-github-token-B1');
    if (!(await page.locator('#sync-on-save').isChecked())) await page.locator('#sync-on-save').check();
    const before = await bodyText(page);
    await page.getByRole('button', { name: 'Save Configuration' }).click(); await wait(4000);
    facts.toasts = await toastLines(before);
    facts.view = await view('github-like');
    facts.fileEntry = mcpsJson(inst).mcpServers['github-like'];
    const g = await waitTool('gh_report_env_and_cwd', { name: 'GITHUB_TOKEN' });
    facts.gatewayDeliveredB1 = g.text ? JSON.parse(g.text).value === 'test-dummy-github-token-B1' : g;
  }
  if (phase === 'reimport') {
    // Re-import the older copy of this install's own file: the server keeps its current, working secret (AR-016).
    const refBefore = mcpsJson(inst).mcpServers['github-like']?.env?.GITHUB_TOKEN;
    await openMcp(page);
    await page.getByRole('button', { name: 'Bulk Import', exact: true }).click(); await page.waitForSelector('#bulk-json-input');
    await page.fill('#bulk-json-input', fs.readFileSync(extra, 'utf8'));
    const before = await bodyText(page);
    await page.getByRole('button', { name: 'Import Configurations' }).click(); await wait(3000);
    facts.toasts = await toastLines(before);
    facts.olderCopyRef = JSON.parse(fs.readFileSync(extra, 'utf8')).mcpServers['github-like']?.env?.GITHUB_TOKEN;
    facts.refBefore = refBefore;
    facts.refAfter = mcpsJson(inst).mcpServers['github-like']?.env?.GITHUB_TOKEN;
    facts.view = await view('github-like');
    await gql(inst, 'mutation($id: String!) { discoverAndRegisterMcpServerTools(serverId: $id) { success message } }', { id: 'github-like' });
    const g = await waitTool('gh_report_env_and_cwd', { name: 'GITHUB_TOKEN' });
    facts.gatewayStillB1 = g.text ? JSON.parse(g.text).value === 'test-dummy-github-token-B1' : g;
    await shot(page, out, 'API-05-04-after-reimport-older-copy');
