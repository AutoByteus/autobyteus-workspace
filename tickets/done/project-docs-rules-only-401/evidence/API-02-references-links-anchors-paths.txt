]


def git(*args):
    return subprocess.run(["git", *args], capture_output=True, text=True).stdout


def slugs(path):
    out, seen, fence = set(), {}, False
    for line in open(path, encoding="utf-8"):
        if line.startswith("```"):
            fence = not fence
        if fence:
            continue
        m = re.match(r"^(#{1,6})\s+(.*?)\s*#*\s*$", line)
        if not m:
            continue
        t = re.sub(r"[^\w\- ]", "", m.group(2).strip().lower()).replace(" ", "-")
        n = seen.get(t, 0)
        seen[t] = n + 1
        out.add(t if n == 0 else f"{t}-{n}")
    return out


def main():
    print("## git grep -n -i 'case studies|#case-studies' in every tracked .md outside tickets/")
    hits = git("grep", "-n", "-i", "-e", "case studies", "-e", "#case-studies", "--", "*.md", ":!tickets/**", ":!**/tickets/**").strip()
    print(hits or "(none)")
    print(f"Case Studies hits: {len(hits.splitlines()) if hits else 0}")

    print("\n## local links and anchors in RELEASE.md, DESIGN.md, TESTING.md and the three records")
    tracked = set(git("ls-files").splitlines())
    total = broken = 0
    for f in FILES:
        text = re.sub(r"```.*?```", "", open(f, encoding="utf-8").read(), flags=re.S)
        n_file = 0
        for m in re.finditer(r"\]\(([^)\s]+)\)", text):
            tgt = m.group(1)
            if re.match(r"[a-z]+:", tgt):
                continue
            total += 1
            n_file += 1
            p, _, anchor = tgt.partition("#")
            dest = os.path.normpath(os.path.join(os.path.dirname(f), p)) if p else f
            ok = dest in tracked or os.path.isdir(dest)
            aok = (anchor.lower() in slugs(dest)) if (ok and anchor and dest.endswith(".md")) else True
            if not (ok and aok):
                broken += 1
                print(f"  BROKEN {f}: ({tgt}) -> {dest} exists={ok} anchor_ok={aok}")
        print(f"  {f}: {n_file} local links")
    print(f"local links checked: {total}; broken: {broken}")

    print("\n## pointer paths (REQ-005 / AC-005) on origin/personal")
    paths = sorted(set(re.findall(r"tickets/done/[a-z0-9.-]+/release-deployment-report\.md",
                                  open("RELEASE.md").read() + open("DESIGN.md").read())))
    for p in paths:
        on = subprocess.run(["git", "cat-file", "-e", f"origin/personal:{p}"]).returncode == 0
        where = " ".join(":".join(l.split(":")[:2]) for l in git("grep", "-n", "-F", p, "--", "RELEASE.md", "DESIGN.md").splitlines())
        print(f"{p}: on origin/personal: {'yes' if on else 'NO'}; referenced at: {where}")

    print("\n## AC-001: version strings left in RELEASE.md")
    print(git("grep", "-n", "-o", r"v[0-9]\+\.[0-9]\+\.[0-9]\+[-a-z0-9.]*", "--", "RELEASE.md").strip() or "(none)")
    print("## AC-001: dates (20YY-MM-DD) left in RELEASE.md")
    print(git("grep", "-n", "20[0-9][0-9]-[01][0-9]-[0-3][0-9]", "--", "RELEASE.md").strip() or "(none)")

    print("\n## REQ-004: TESTING.md unchanged by this ticket")
    print(f"git diff --stat {sys.argv[1]} HEAD -- TESTING.md: '{git('diff', '--stat', sys.argv[1], 'HEAD', '--', 'TESTING.md').strip()}'")


main()
