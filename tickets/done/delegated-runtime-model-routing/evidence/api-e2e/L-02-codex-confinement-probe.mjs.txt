#!/usr/bin/env node
// autobyteus-32 live router probe (temporary; a copy is kept in the ticket's evidence). A built node on a fresh data root,
// the real HOME (the CLIs' sign-ins; nothing in ~/.claude, ~/.codex, ~/.grok is written) and a throwaway CODEX_HOME whose
// auth.json links the real one. Phases, in order, on one node:
//   L-01 the Model Router on Claude: tool list (no shell), writes outside its workspace (a $TMPDIR path and a HOME path)
//        denied although the workspace's .claude/settings.json allows everything; a write inside is allowed;
//   L-03 AC-007: the owner teaches the router "writing → Antigravity" in its own run; knowledge.md changes; the next
//        writing Task from a live delegator (Claude haiku) is routed to Antigravity, with the reason;
//   L-04 C-05: the router's CLI process is killed; two more routings are observed (result, notes, time, status);
//   L-02 the Model Router on Codex: writes outside the workspace refused although the workspace's .codex/config.toml asks
//        for danger-full-access; inside allowed.
// Usage: node router-live-probe.mjs <worktree> <out-dir>
import { spawn, execFileSync } from 'node:child_process';
import { createWriteStream, existsSync } from 'node:fs';
import fs from 'node:fs/promises';
import net from 'node:net';
import os from 'node:os';
import path from 'node:path';
import { randomUUID } from 'node:crypto';
import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';

const [worktree, outDir] = process.argv.slice(2);
const PHASES = (process.env.MR_PHASES ?? 'L-01,L-03,L-04,L-02').split(',');
const serverDir = path.join(worktree, 'autobyteus-server-ts');
const WebSocket = createRequire(path.join(serverDir, 'package.json'))('ws');
const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
const out = { startedAt: new Date().toISOString(), head: execFileSync('git', ['-C', worktree, 'rev-parse', '--short', 'HEAD']).toString().trim(), phases: {} };
await fs.mkdir(outDir, { recursive: true });
const save = () => fs.writeFile(path.join(outDir, 'router-live-probe.json'), JSON.stringify(out, null, 2));
const port = await new Promise((resolve) => { const s = net.createServer(); s.listen(0, '127.0.0.1', () => { const { port } = s.address(); s.close(() => resolve(port)); }); });
const root = await fs.realpath(await fs.mkdtemp(path.join(os.tmpdir(), 'mr-live-')));
const dataRoot = path.join(root, 'node');
// Outside targets: one under $TMPDIR, one in HOME (Codex's workspace-write may write its temp roots by design).
const homeOutside = path.join(os.homedir(), `.mr-confinement-probe-${randomUUID().slice(0, 8)}`);
const tmpOutside = path.join(root, 'outside');
for (const dir of [homeOutside, tmpOutside]) await fs.mkdir(dir, { recursive: true });
for (const folder of ['db', 'logs', 'memory', 'temp_workspace', 'workspace']) await fs.mkdir(path.join(dataRoot, folder), { recursive: true });
const codexHome = path.join(root, 'codex-home');
await fs.mkdir(codexHome, { recursive: true });
await fs.symlink(path.join(os.homedir(), '.codex', 'auth.json'), path.join(codexHome, 'auth.json'));
const url = `http://127.0.0.1:${port}`;
const db = pathToFileURL(path.join(dataRoot, 'db', 'node.db')).href;
const env = { ...Object.fromEntries(Object.entries(process.env).filter(([k]) => !k.startsWith('AUTOBYTEUS_') && !k.startsWith('ENABLE_') && !/_API_KEY$/.test(k))),
  CODEX_HOME: codexHome, CLAUDE_SESSION_RAW_EVENT_DEBUG: '1', CLAUDE_SESSION_RAW_EVENT_MAX_CHARS: '200000', APP_ENV: 'development', DB_TYPE: 'sqlite', DATABASE_URL: db, AUTOBYTEUS_SERVER_HOST: url, AUTOBYTEUS_LOG_DIR: path.join(dataRoot, 'logs'),
  AUTOBYTEUS_MEMORY_DIR: path.join(dataRoot, 'memory'), AUTOBYTEUS_TEMP_WORKSPACE_DIR: path.join(dataRoot, 'temp_workspace') };
await fs.writeFile(path.join(dataRoot, '.env'), `APP_ENV=development\nDB_TYPE=sqlite\nDATABASE_URL=${db}\nAUTOBYTEUS_SERVER_HOST=${url}\n`);
execFileSync('corepack', ['pnpm', '-C', serverDir, 'exec', 'prisma', 'migrate', 'deploy', '--schema', './prisma/schema.prisma'], { env: { ...env, HOME: os.homedir() }, stdio: 'ignore' });
const log = createWriteStream(path.join(outDir, 'router-live-probe-server.log'));
const server = spawn(process.execPath, [path.join(serverDir, 'dist/app.js'), '--host', '127.0.0.1', '--port', String(port), '--data-dir', dataRoot], { cwd: serverDir, env, detached: true, stdio: ['ignore', 'pipe', 'pipe'] });
server.stdout.pipe(log); server.stderr.pipe(log);
const gql = async (query, variables = {}) => { const r = await fetch(`${url}/graphql`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ query, variables }) }).then((x) => x.json()); if (r.errors) throw new Error(JSON.stringify(r.errors)); return r.data; };
const sockets = [];
const connect = async (runId) => {
  const socket = new WebSocket(`ws://127.0.0.1:${port}/ws/agent/${runId}`);
  sockets.push(socket);
  const frames = []; socket.on('message', (raw) => frames.push({ ...JSON.parse(String(raw)), at: Date.now() }));
  await new Promise((resolve, reject) => { socket.once('open', resolve); socket.once('error', reject); });
  for (let i = 0; i < 600 && !frames.some((f) => f.type === 'CONNECTED'); i += 1) await sleep(100);
  return { socket, frames, runId };
};
/** One turn: sends `content`, waits for idle after the turn's own activity; returns the turn's frames and text. */
const turn = async (channel, content, ms = 240_000) => {
  const from = channel.frames.length;
  const id = `mr-${randomUUID()}`;
  const t0 = Date.now();
  channel.socket.send(JSON.stringify({ type: 'SEND_MESSAGE', payload: { agent_run_id: channel.runId, content, message_id: id, dedupe_key: `agent_run_input:e2e:${id}`, context_file_paths: [], image_urls: [] } }));
  let busy = false;
  while (Date.now() - t0 < ms) {
    const slice = channel.frames.slice(from);
    busy ||= slice.some((f) => f.type === 'AGENT_STATUS' && f.payload?.status !== 'idle') || slice.some((f) => f.type.startsWith('SEGMENT_') || f.type.startsWith('TOOL_'));
    const lastStatus = [...slice].reverse().find((f) => f.type === 'AGENT_STATUS');
    if (busy && lastStatus?.payload?.status === 'idle' && Date.now() - lastStatus.at > 1500) break;
    await sleep(500);
  }
  const slice = channel.frames.slice(from);
  return { ms: Date.now() - t0,
    tools: slice.filter((f) => f.type.startsWith('TOOL_')).map((f) => ({ type: f.type, tool: f.payload?.tool_name, args: JSON.stringify(f.payload?.arguments ?? null).slice(0, 300),
      result: JSON.stringify(f.payload?.result ?? f.payload?.error ?? null).slice(0, 400), at: f.at - t0 })),
    text: slice.filter((f) => f.type === 'SEGMENT_CONTENT').map((f) => String(f.payload?.delta ?? f.payload?.content ?? '')).join('').slice(-1500),
    errors: slice.filter((f) => /ERROR/.test(f.type)).map((f) => JSON.stringify(f.payload).slice(0, 400)) };
};
const exists = (file) => existsSync(file);
const report = async () => (await gql('{ availableAgentRuntimes }')).availableAgentRuntimes;
const modelsOf = (rep, runtime) => (rep.agentRuntimes.find((e) => e.runtime === runtime)?.models ?? []).filter((m) => !m.blocked).map((m) => m.id);
const metadataOf = async (runId) => JSON.parse(await fs.readFile(path.join(dataRoot, 'memory', 'agents', runId, 'run_metadata.json'), 'utf8'));
const routing = {
  router: (r, m) => gql('mutation($r:String!,$m:String!){setModelRouterLaunch(runtimeKind:$r, llmModelIdentifier:$m)}', { r, m }).then((d) => d.setModelRouterLaunch),
  enable: (e) => gql('mutation($e:Boolean!){setModelRoutingEnabled(enabled:$e)}', { e }).then((d) => d.setModelRoutingEnabled),
  status: () => gql('{ modelRoutingStatus }').then((d) => d.modelRoutingStatus),
  open: () => gql('mutation{ openModelRouterAgent }').then((d) => d.openModelRouterAgent),
  decisions: () => gql('{ modelRoutingDecisions(limit: 50) }').then((d) => d.modelRoutingDecisions),
};
const segment = (name) => name.toLowerCase().replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '');
const createDefinition = async (name, instructions, extra = {}) => (await gql('mutation($i:CreateAgentDefinitionInput!){createAgentDefinition(input:$i){id}}',
  { i: { name, role: 'assistant', description: extra.description ?? `${name} (autobyteus-32 live probe)`, instructions, toolNames: [], ...extra.input } })).createAgentDefinition.id;
const createRun = async (definitionId, runtimeKind, model) => {
  const created = (await gql('mutation($i:CreateAgentRunInput!){createAgentRun(input:$i){success message runId}}', { i: { agentDefinitionId: definitionId,
    workspaceRootPath: path.join(dataRoot, 'workspace'), llmModelIdentifier: model, llmConfig: null, autoExecuteTools: true, runtimeKind } })).createAgentRun;
  if (!created.success) throw new Error(created.message);
  return created.runId;
};
/** Processes whose working directory is `dir` and that descend from the server. */
const processesIn = (dir) => {
  const lsof = (() => { try { return execFileSync('lsof', ['-a', '-d', 'cwd', '-Fpn', '+c', '0'], { encoding: 'utf8' }); } catch (e) { return e.stdout ?? ''; } })();
  const pids = []; let pid = null;
  for (const line of lsof.split('\n')) { if (line.startsWith('p')) pid = Number(line.slice(1)); else if (line.startsWith('n') && line.slice(1) === dir) pids.push(pid); }
  const parentOf = (p) => { try { return Number(execFileSync('ps', ['-o', 'ppid=', '-p', String(p)], { encoding: 'utf8' }).trim()); } catch { return 0; } };
  const descends = (p) => { for (let i = 0; i < 12 && p > 1; i += 1) { if (p === server.pid) return true; p = parentOf(p); } return false; };
  return pids.filter(descends).map((p) => ({ pid: p, command: (() => { try { return execFileSync('ps', ['-o', 'command=', '-p', String(p)], { encoding: 'utf8' }).trim().slice(0, 200); } catch { return ''; } })() }));
};

try {
  while (!(await fetch(`${url}/rest/health`).then((r) => r.ok).catch(() => false))) { if (server.exitCode !== null) throw new Error('server exited'); await sleep(250); }
  const codexModels = modelsOf(await report(), 'codex_app_server');
  out.runtimes = { codex: codexModels };
  const codexPolicies = async () => {
    const files = [];
    const walk = async (dir) => { for (const e of await fs.readdir(dir, { withFileTypes: true }).catch(() => [])) { const f = path.join(dir, e.name); if (e.isDirectory()) await walk(f); else if (f.endsWith('.jsonl')) files.push(f); } };
    await walk(path.join(codexHome, 'sessions'));
    const policies = [];
    for (const file of files) for (const line of (await fs.readFile(file, 'utf8')).split('\n').filter(Boolean)) {
      let entry; try { entry = JSON.parse(line); } catch { continue; }
      if (entry.type === 'turn_context') policies.push({ file: path.basename(file), approval_policy: entry.payload?.approval_policy, sandbox_policy: entry.payload?.sandbox_policy });
    }
    return policies;
  };
  const target = path.join(homeOutside, 'codex-script-write.txt');
  const phase = async (label, model, withConfig) => {
    // The router's internal workspace (written before the router run, and so before Codex first starts on this node).
    const workspace = path.join(dataRoot, 'internal-workspaces', 'autobyteus-model-router');
    await fs.mkdir(workspace, { recursive: true });
    await fs.rm(path.join(workspace, '.codex'), { recursive: true, force: true });
    if (withConfig) {
      await fs.mkdir(path.join(workspace, '.codex'), { recursive: true });
      await fs.writeFile(path.join(workspace, '.codex', 'config.toml'), 'sandbox_mode = "danger-full-access"\napproval_policy = "never"\n[sandbox_workspace_write]\nwritable_roots = ["/"]\n');
    }
    await routing.router('codex_app_server', model);
    await routing.enable(true);
    const runId = await routing.open();
    if (await fs.realpath((await metadataOf(runId)).workspaceRootPath) !== await fs.realpath(workspace)) throw new Error('unexpected router workspace');
    // A workspace script whose output file is named in a data file (the sandbox, not the model, must refuse the write).
    await fs.writeFile(path.join(workspace, 'target.txt'), `${target}\n`);
    await fs.writeFile(path.join(workspace, 'probe.sh'), '#!/bin/sh\nP="$(cat ./target.txt)"\nif echo SANDBOX-TEST > "$P" 2>/tmp/probe.err; then echo "WROTE"; else echo "REFUSED: $(cat /tmp/probe.err)"; fi\n');
    await fs.rm(target, { force: true });
    const channel = await connect(runId);
    const t = await turn(channel, 'Node owner check, not a routing request: in your workspace run exactly `sh probe.sh` once and reply with its output verbatim. Do not run anything else.');
    out.phases[label] = { runId, model, withConfig, turn: t, targetWritten: exists(target), targetContent: exists(target) ? await fs.readFile(target, 'utf8') : null };
    await fs.rm(target, { force: true });
    await save();
  };
  // With the workspace config present from the start, then (a new run on another launch) without it.
  await phase('codex-with-workspace-config', codexModels[0], true);
  await phase('codex-control-after', codexModels[1] ?? codexModels[0], false);
  out.phases.codexPolicies = await codexPolicies();
  await routing.enable(false);
} catch (error) {
  out.error = error.stack || String(error);
} finally {
  for (const socket of sockets) { try { socket.close(); } catch {} }
  try { process.kill(-server.pid, 'SIGTERM'); } catch {}
  await sleep(4000);
  try { process.kill(-server.pid, 'SIGKILL'); } catch {}
  out.cleanup = { serverExited: server.exitCode !== null || server.signalCode !== null };
  await fs.rm(root, { recursive: true, force: true });
  await fs.rm(homeOutside, { recursive: true, force: true });
  out.cleanup.removed = { tempRoot: !existsSync(root), homeProbeDir: !existsSync(homeOutside) };
  out.finishedAt = new Date().toISOString();
  await save();
  console.log(JSON.stringify({ error: out.error ?? null, phases: Object.keys(out.phases), cleanup: out.cleanup }));
}
