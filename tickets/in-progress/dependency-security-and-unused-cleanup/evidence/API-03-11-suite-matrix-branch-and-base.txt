== runner (run-suites.sh)
#!/bin/bash
# API/E2E (autobyteus-229): build + full suite rerun for one side. Usage: run-suites.sh <worktree> <side>
# Each step: /tmp/dep229v/logs/<side>-<step>.log (header + full output, ANSI stripped); summary: /tmp/dep229v/logs/<side>-summary.txt
W="$1"; side="$2"; L=/tmp/dep229v/logs; S=$L/$side-summary.txt
cd "$W" || exit 9
echo "# $side commit=$(git rev-parse --short HEAD) start=$(date -u +%FT%TZ)" > $S
step() { name="$1"; shift; f=$L/$side-$name.log; echo "# $name ($side, commit $(git rev-parse --short HEAD), $(date -u +%FT%TZ)); cmd: $*" > $f; t0=$(date +%s); bash -c "$*" 2>&1 | sed -e 's/\x1b\[[0-9;]*[A-Za-z]//g' >> $f; rc=${PIPESTATUS[0]}; echo "$name exit=$rc secs=$(( $(date +%s)-t0 )) $(grep -E '^ *(Tests|Test Files) |# (tests|pass|fail) |Lockfile is up to date|ERR_PNPM|\[verify:runtime-deps\]' $f | tr -s ' ' | tr '\n' ';' | cut -c1-400)" >> $S; }
H() { mktemp -d | xargs -I{} sh -c 'cd {} && pwd -P'; }
step install "pnpm install --frozen-lockfile"
step dedupe-check "pnpm dedupe --check"
step server-prebuild "pnpm -C autobyteus-server-ts prebuild"
for p in autobyteus-application-sdk-contracts autobyteus-team-stream-contracts autobyteus-agent-presentation-contracts autobyteus-collaboration-stream-contracts autobyteus-application-frontend-sdk autobyteus-application-backend-sdk autobyteus-application-devkit; do step "pkg-$p" "pnpm -C $p build && pnpm -C $p test"; done
step ts-build "pnpm -C autobyteus-ts build"
step ts-test "env -i PATH=\"$PATH\" HOME=\"$(H)\" TMPDIR=\"$TMPDIR\" pnpm -C autobyteus-ts test --run"
step server-build "pnpm -C autobyteus-server-ts build"
step server-typecheck "pnpm -C autobyteus-server-ts typecheck"
step server-unit "pnpm -C autobyteus-server-ts test:unit"
step server-int "pnpm -C autobyteus-server-ts test:integration:prepare && pnpm -C autobyteus-server-ts test:integration"
step server-e2e "env -i PATH=\"$PATH\" HOME=\"$(H)\" TMPDIR=\"$TMPDIR\" pnpm test:e2e"
step web-prepare "pnpm -C autobyteus-web exec nuxt prepare"
step web-guards "pnpm -C autobyteus-web guard:web-boundary && pnpm -C autobyteus-web guard:localization-boundary && pnpm -C autobyteus-web audit:localization-literals"
step web-build "pnpm -C autobyteus-web build"
step web-test-nuxt "pnpm -C autobyteus-web test:nuxt --run"
step web-test-electron "pnpm -C autobyteus-web test:electron --run"
step web-voice-input "NUXT_TEST=true pnpm -C autobyteus-web exec vitest run tests/integration/voice-input-extension.integration.test.ts"
for a in brief-studio socratic-math-teacher; do step "app-$a" "pnpm -C applications/$a typecheck:backend && pnpm -C applications/$a build"; done
echo "# end=$(date -u +%FT%TZ)" >> $S
