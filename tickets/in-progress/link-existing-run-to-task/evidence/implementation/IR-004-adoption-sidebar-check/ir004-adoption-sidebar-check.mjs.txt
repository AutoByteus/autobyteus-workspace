const until = async (label, predicate, ms = 30000) => {

const exited = (c) => !c || c.exitCode !== null || c.signalCode !== null;
const stopGroup = async (child) => {
  if (exited(child)) return { pid: child?.pid ?? null, status: 'not-running' };
  process.kill(-child.pid, 'SIGTERM');
  await until('owned process exit', () => exited(child), 15000).catch(async () => {
    process.kill(-child.pid, 'SIGKILL'); await until('owned process killed', () => exited(child), 5000);
  });
  return { pid: child.pid, status: 'terminated', exit: child.exitCode ?? child.signalCode };
};
const spawnLogged = (cmd, args, cwd, env, name) => {
  const log = fs.createWriteStream(path.join(outputDir, `${name}.log`), { flags: 'a' });
  const child = spawn(cmd, args, { cwd, env, detached: true, stdio: ['ignore', 'pipe', 'pipe'] });
  child.output = '';
  child.stdout.on('data', (d) => { child.output += d; log.write(d); });
  child.stderr.on('data', (d) => { child.output += d; log.write(d); });
  child.once('close', () => log.end());
  return child;
};

/** Task execution nodes of a stored tree (camelCase Agent/Org trees, snake_case Team tree). */
const taskNodes = (tree) => {
  const found = [];
  const visit = (v) => {
    if (Array.isArray(v)) return v.forEach(visit);
    if (!v || typeof v !== 'object') return;
    const startedAt = v.startedAt ?? v.started_at; const agentRunId = v.agentRunId ?? v.agent_run_id; const teamRunId = v.teamRunId ?? v.team_run_id;
    if (typeof startedAt === 'string' && (agentRunId || teamRunId)) found.push({ agentRunId: teamRunId ? undefined : agentRunId, teamRunId, address: v.address,
      delegatorAgentRunId: v.delegatorAgentRunId ?? v.delegator_agent_run_id, members: (v.members ?? []).map((m) => m.agentRunId ?? m.agent_run_id).filter(Boolean) });
    Object.values(v).forEach(visit);
  };
  visit(tree); return found;
};

const sockets = [];
/** The Manager's input channel (what its composer sends). Used for the Manager's own steps. */
const managerInput = async (root) => {
  const route = root.kind === 'agent' ? `agent/${root.rootId}` : root.kind === 'team' ? `agent-team/${root.rootId}` : `agent-org/${root.rootId}`;
  const socket = new WebSocket(`${stack.backendUrl.replace(/^http/, 'ws')}/ws/${route}`); sockets.push(socket);
  await new Promise((resolve, reject) => { socket.once('open', resolve); socket.once('error', reject); });
  const send = (content) => socket.send(JSON.stringify({ type: 'SEND_MESSAGE', payload: root.kind === 'org'
    ? { root_subject_kind: 'agent_org', root_run_id: root.rootId, target_agent_run_id: root.managerRunId, command_id: randomUUID(),
      message_id: randomUUID(), dedupe_key: randomUUID(), content, context_file_paths: [], image_urls: [] }
    : { agent_run_id: root.managerRunId, message_id: `e2e-${randomUUID()}`, dedupe_key: `agent_run_input:e2e:${randomUUID()}`,
      content, context_file_paths: [], image_urls: [] } }));
  return { send, close: () => socket.close() };
};

/** A root with Task A (Agent + delegated helper + the helper's own delegation to a researcher), Task B (a Team), and non-Task work. */
const setupRoot = async (kind, label, { withB = true, withPlain = true } = {}) => {
  const { ids, names } = await createDefinitions(label);
  const { projectId, task } = await createProject(label);
  const at = (member, catalogName) => (kind !== 'org' ? `/${segment(catalogName)}` : `/${member}`);
  const taskA = await task(callTool('delegate_task', { recipient_address: at('helper', names.helper),
    description: callTool('delegate_task', { recipient_address: `/${segment(names.assistant)}`, description: 'Collect the changelog links.' }) }));
  const taskB = await task('Review the docs site.');
  const root = await createRoot(kind, ids);
  const input = await managerInput(root);
  const count = async () => taskNodes(await storedTree(root)).length;
  input.send(callTool('delegate_task', { recipient_address: at('worker', names.worker), task_id: taskA }));
  await until('Task A runs', async () => (await count()) >= 3, 60000);
  const aNodes = taskNodes(await storedTree(root));
  if (withB) { input.send(callTool('delegate_task', { recipient_address: kind === 'org' ? '/squad' : `/${segment(names.squad)}`, task_id: taskB })); await until('Task B', async () => (await count()) >= 4, 60000); }
  if (withPlain) { input.send(callTool('delegate_task', { recipient_address: at('worker', names.worker), description: 'Draft a short summary.' }));
    await until('non-Task run', async () => (await count()) >= aNodes.length + (withB ? 2 : 1), 60000); }
  const all = taskNodes(await storedTree(root));
  const key = (n) => n.teamRunId ?? n.agentRunId;
  const bNode = all.find((n) => n.teamRunId && !aNodes.some((a) => key(a) === key(n)));
  const plainNode = all.find((n) => !n.teamRunId && !aNodes.some((a) => key(a) === key(n)));
  return { ids, names, projectId, taskA, taskB, root, input, at,
    aRefs: aNodes.map((n) => (n.teamRunId ? { teamRunId: n.teamRunId } : { agentRunId: n.agentRunId })),
    aAssigned: aNodes.find((n) => n.delegatorAgentRunId === root.managerRunId),
    bRef: bNode ? { teamRunId: bNode.teamRunId } : null, bMembers: bNode?.members ?? [], plainRef: plainNode ? { agentRunId: plainNode.agentRunId } : null };
};

const journey = async () => {
  const out = {};
  const { ids, names } = await createDefinitions('Adopt');
  const { projectId, task } = await createProject('Adopt');
  const agentTask = await task('Write the release notes.');
  const teamTask = await task('Review the docs site.');
  // Runs the user started by hand: an agent run and a team run, each with its own conversation.
  const worker = (await gql('mutation($input:CreateAgentRunInput!){createAgentRun(input:$input){success message runId}}',
    { input: { agentDefinitionId: ids.worker, ...config() } })).createAgentRun.runId;
  const workerInput = await managerInput({ kind: 'agent', rootId: worker, managerRunId: worker });
  workerInput.send('Hello, start of my own work.');
  await until('worker answered', async () => (await conversationText(worker)).includes('OK'), 60000);
  const teamRun = (await gql('mutation($input:CreateAgentTeamRunInput!){createAgentTeamRun(input:$input){success message teamRunId}}',
    { input: { teamDefinitionId: ids.squad, teamConfigs: [{ teamAddress: '/', ...config() }], memberConfigs: [['reviewer', ids.worker], ['editor', ids.helper]]
      .map(([m, id]) => ({ memberAddress: `/${m}`, agentDefinitionId: id, ...config() })) } })).createAgentTeamRun;
  assert(teamRun.teamRunId, teamRun.message);
  const team = teamRun.teamRunId;
  const tree = (await gql('query($id:String!){getTeamRunResumeConfig(teamRunId:$id){executionTree}}', { id: team })).getTeamRunResumeConfig.executionTree;
  const reviewer = tree.root_team.members.find((m) => m.address === '/reviewer').agent_run_id;
  const teamInput = await managerInput({ kind: 'team', rootId: team, managerRunId: reviewer });
  teamInput.send('Hello team, start of our own work.');
  await until('team answered', async () => (await conversationText(reviewer)).includes('OK'), 60000).catch(() => sleep(10000));
  workerInput.close(); teamInput.close();
  const pm = await createRoot('agent', ids);
  const pmInput = await managerInput(pm);
  pmInput.send('Hello, I manage the project.');
  await until('PM answered', async () => (await conversationText(pm.rootId)).includes('OK'), 60000);

  // Page 1: the user opened the hand-started agent run from its history row, and the PM run (delegator open).
  const p1 = await newPage();
  await openWorkspace(p1.page);
  await openAgentRun(p1.page, names.worker, worker);
  await openAgentRun(p1.page, names.manager, pm.rootId);
  await shot(p1.page, '01-before-adoption');
  out.beforeAdoption = { ownRow: await count(p1.page, ownAgentRow(worker)) };
  assert(out.beforeAdoption.ownRow === 1, 'the opened worker run is not in its own group before adoption');

  pmInput.send(callTool('delegate_task', { target_agent_run_id: worker, task_id: agentTask }));
  await until('agent run adopted (stored tree)', async () => taskNodes(await storedTree(pm)).some((n) => n.agentRunId === worker), 90000);
  const adoptedAt = Date.now();
  await until('adopted agent run left its own group', async () => (await count(p1.page, ownAgentRow(worker))) === 0, 30000);
  out.agentLeftOwnGroupMs = Date.now() - adoptedAt;
  await until('adopted agent run under the PM', async () => (await count(p1.page, rowSelector({ agentRunId: worker }))) >= 1, 30000);
  out.afterAdoption = { ownRow: await count(p1.page, ownAgentRow(worker)), underPm: await count(p1.page, rowSelector({ agentRunId: worker })),
    retired: await retiredState(p1.page) };
  await shot(p1.page, '02-after-adoption');
  await sleep(6000); // past one history refresh (5 s)
  out.afterHistoryRefresh = { ownRow: await count(p1.page, ownAgentRow(worker)), retired: await retiredState(p1.page) };
  assert(out.afterHistoryRefresh.ownRow === 0, 'the adopted run came back in its own group after a history read');

  // AC-003 / D-1: DONE removes it from under the PM; reopening and a message bring it back there, never to its own group.
  pmInput.send(callTool('update_task', { task_id: agentTask, status: 'DONE' }));
  await until('Task DONE', async () => (await taskStatus(projectId, agentTask)) === 'DONE', 60000);
  await until('row under the PM leaves on DONE', async () => (await count(p1.page, rowSelector({ agentRunId: worker }))) === 0, 30000);
  await sleep(6000);
  out.afterDone = { ownRow: await count(p1.page, ownAgentRow(worker)), underPm: await count(p1.page, rowSelector({ agentRunId: worker })) };
  await shot(p1.page, '03-after-done');
  assert(out.afterDone.ownRow === 0, 'the adopted run came back in its own group after DONE');
  pmInput.send(callTool('update_task', { task_id: agentTask, status: 'IN_PROGRESS' }));
  await until('Task reopened', async () => (await taskStatus(projectId, agentTask)) === 'IN_PROGRESS', 60000);
  pmInput.send(callTool('send_message_to', { target_agent_run_id: worker, content: 'Continue the Task please.' }));
  await until('reactivated under the PM', async () => (await count(p1.page, rowSelector({ agentRunId: worker }))) >= 1, 60000);
  await sleep(6000);
  out.afterReopen = { ownRow: await count(p1.page, ownAgentRow(worker)), underPm: await count(p1.page, rowSelector({ agentRunId: worker })) };
  await shot(p1.page, '04-after-reopen');
  assert(out.afterReopen.ownRow === 0, 'the adopted run came back in its own group after reopening');

  // Page 2: the user opened the hand-started team run only (the PM is not opened in this page).
  const p2 = await newPage();
  await openWorkspace(p2.page);
  await p2.page.locator(`[data-test="workspace-team-definition-row-${slug(names.squad)}"]`).click();
  await until('team row', async () => (await count(p2.page, ownTeamRow(team))) === 1, 30000);
  await p2.page.locator(ownTeamRow(team)).click();
  await until('team tree', async () => (await count(p2.page, '[data-test="workspace-team-execution-tree"]')) >= 1, 30000);
  await shot(p2.page, '05-team-before-adoption');
  pmInput.send(callTool('delegate_task', { target_team_run_id: team, task_id: teamTask }));
  await until('team run adopted (stored tree)', async () => taskNodes(await storedTree(pm)).some((n) => n.teamRunId === team), 90000);
  const teamAdoptedAt = Date.now();
  await until('adopted team left its own group', async () => (await count(p2.page, ownTeamRow(team))) === 0, 30000);
  out.teamLeftOwnGroupMs = Date.now() - teamAdoptedAt;
  out.teamRetired = await retiredState(p2.page);
  await shot(p2.page, '06-team-after-adoption');

  // A reload: the server lists neither run; both are shown only under the PM.
  await openWorkspace(p1.page);
  await sleep(3000);
  out.afterReload = { workerOwnRow: await count(p1.page, ownAgentRow(worker)), teamOwnRow: await count(p1.page, ownTeamRow(team)) };
  assert(out.afterReload.workerOwnRow === 0 && out.afterReload.teamOwnRow === 0, 'an adopted run is in its own group after reload');
  out.browserErrors = [...p1.errors, ...p2.errors];
  assert(out.browserErrors.length === 0, 'browser errors', out.browserErrors);
  return out;
};
