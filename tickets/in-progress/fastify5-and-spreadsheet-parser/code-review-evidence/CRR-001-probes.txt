app.get("/rest/health", async () => ({ ok: 1 }));

const pick = (h) => Object.fromEntries(Object.entries(h).filter(([k]) => k.startsWith("access-control") || k === "vary").sort());
const cases = [

  ["POST", "/graphql", { origin: "http://localhost:3000", "content-type": "application/json" }],
  ["GET", "/rest/health", {}],
];
for (const [method, url, headers] of cases) {

await app.close();
